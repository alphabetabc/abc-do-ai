```js
// 新增工具函数（不 cloneDeep，纯递归查找）
export function getFieldNodeById(components, uniqueId) {
    for (const node of components) {
        if (node.uniqueId === uniqueId) return node;
        if (node.children) {
            const found = getFieldNodeById(node.children, uniqueId);
            if (found) return found;
        }
    }
    return null;
}
```

### 3.2 getFieldOrderBy 改写

旧：

```js
function getFieldOrderBy(fields, id, opts = {}) {
    const { rebuild = true } = opts;
    let newFiled = rebuild ? _.cloneDeep(fields) : fields;
    const index = newFiled.findIndex((o) => o.uniqueId === id);
    return { index, components: newFiled };
}
```

- `rebuild=true` 时 cloneDeep 整个 children（layer-manager 全部用 rebuild=true）
- 双层 cloneDeep（getParent 已经 cloneDeep 过一次）

```js

 * 调用方模式：
 *   const parentId = parentMap[uniqueId];
 *   const parent = byId[parentId];            // 父组件 FlatField
 *   const index = getIndexById(parentMap, byId, uniqueId, parent);  // 在父 children 中的位置
 */
function getIndexById(parentMap, byId, uniqueId, parent) {
    // 遍历 components 树找到该 child 的位置
    // 改为：由 components 树直接 findIndex（仍是 O(n)，但不需要 cloneDeep）
    // 因为 components 是稳定引用（Immer 不会无故重建），findIndex 安全
}
```

**实际**：保留 `getFieldOrderBy` 函数本身，但内部从 `_.cloneDeep(fields)` 改为**不 cloneDeep**——因为 `components` 是不可变替换的，外部不会意外修改：

```js
export function getFieldOrderBy(fields, id) {
    if (isEmpty(fields)) return {};
    const index = fields.findIndex((o) => o.uniqueId === id);
    return { index, components: fields };
}
```

#### ⚠️ 关键风险：layer-manager splice 崩溃

`layer-manager/move/index.ts` 现状（[L16-21](src/designer/layer-manager/move/index.ts#L16)）：

```js
const { index, components } = getFieldOrderBy(parents.children, selected);
const temp = components.splice(index, 1)[0];   // ← 直接 mutate 返回的 components
components.unshift(temp);                       // ← 同上
```

**去掉 cloneDeep 后**，`getFieldOrderBy` 返回的 `components` 就是 `parents.children` 的原引用（`getParent` 也不 cloneDeep 了）。`splice` / `unshift` 会直接 mutate `state.components` 的嵌套数组。**搬到 Redux（Immer）后，Immer 代理会检测到 mutation 并抛 `TypeError: Cannot assign to read only property`**。

**改造方案**：layer-manager 的 splice/unshift/push 必须改为不可变写法：

```js
// 旧（moveToTop）
const temp = components.splice(index, 1)[0];
components.unshift(temp);
const finalData = setChildren(state.components, parents.uniqueId, components);

// 新（moveToTop）—— 不可变
const newChildren = [components[index], ...components.slice(0, index), ...components.slice(index + 1)];
// 注意：moveToTop 是把 index 移到最前，等价于：
const newChildren = [
    components[index],
    ...components.filter((_, i) => i !== index),
];
const finalData = setChildren(state.components, parents.uniqueId, newChildren);
```

**需要改造的 layer-manager 函数清单**：

| 函数 | 文件 | 原 mutation | 不可变写法 |
|---|---|---|---|
| `moveToTop` | `move/index.ts` L19-21 | `splice` + `unshift` | `[components[index], ...components.filter((_, i) => i !== index)]` |
| `moveToBottom` | `move/index.ts` L42-44 | `splice` + `push` | `[...components.filter((_, i) => i !== index), components[index]]` |
| `moveIndexToUp` | `move/index.ts` L64 | `orderBy(components, index, index-1)`（splice 交换） | `orderBy` 内部改为不可变（见下） |
| `moveIndexToDown` | `move/index.ts` L85 | 同上 | 同上 |
| `orderBy` | `utils.js` L250-253 | `arr[next] = arr.splice(prev, 1, arr[next])[0]` | 改为 `filter` + `slice` 重组 |
| `deleteFieldByUniqueId` | `utils.js` L273-291 | `components.splice(index, 1)` | `[...components.slice(0, index), ...components.slice(index + 1)]` |

**`orderBy` 不可变改写**：

```js
// 旧
export function orderBy(arr, next, prev) {
    arr[next] = arr.splice(prev, 1, arr[next])[0];
    return arr;
}

// 新（返回新数组，不改原数组）
export function orderBy(arr, next, prev) {
    const result = [...arr];
    const [moved] = result.splice(prev, 1);
    result.splice(next, 0, moved);
    return result;
}
```

**注意**：`orderBy` 改为返回新数组后，所有调用方（`moveIndexToUp` / `moveIndexToDown`）不需要再改——它们已经用 `const results = orderBy(...)` 接收返回值。

**验证**：grep `splice\|unshift\|push\(temp\)` 在 layer-manager + utils.js 应返回 0 命中（除不可变写法内部）。

**替代方案**（更激进）：完全删 `getFieldOrderBy`，改用 `byId` + `parentMap` 推导——但这要求 layer-manager 大改。**本期保守做法：保留函数但去掉 cloneDeep + 同步改 layer-manager splice**。

### 3.3 setLevelPath 改写

> **重要**：task-008 已经提前把 `setLevelPath` 改为不可变（[task-008 §3.4 步骤 3d](./task-2026-07-21-008-patch-field-config.md)），task-009 阶段这里只做收尾确认（grep 校验所有调用方都用返回值）。如果 task-008 已经做对，本节验证即可；如果有遗漏，task-009 补齐。

旧：

```js
export function setLevelPath(nodes, parentNode, key = 'drillDownLevel') {
    if (nodes.length === 0) return nodes;
    for (let i = 0; i < nodes.length; i++) {
        if (!parentNode) {
            nodes[i].data[key] = 0;
        } else {
            nodes[i].data[key] = parentNode.data[key] + 1;
        }
        if (nodes[i].data.config.drillDown && nodes[i].data.config.drillDown.length > 0) {
            setLevelPath(nodes[i].data.config.drillDown, nodes[i], key);
        }
    }
}
```

**问题**：直接修改 nodes（mutation）。

#### ⚠️ 关键风险：drillDown 子节点不在 byId 索引里

`setLevelPath` 不仅遍历 `nodes[i].children`，还遍历 `nodes[i].data.config.drillDown`（轮播组件的层级数据）。但 task-007 §4.1 的 `buildIndex` 只遍历 `node.children`，**不遍历 `data.config.drillDown`**——所以 byId 里**没有 drillDown 子节点的 entry**。

如果 `setLevelPathById` 试图写 `newById[node.uniqueId]`，drillDown 子节点会查不到，导致 level 设置失效或 NPE。

#### 决策：setLevelPath 仍遍历 components 树，但不经过 byId

`drillDownLevel` 是**低频操作**（仅在 `onValueChange` / `lock` / `visible` 等场景调用，不是高频 onChange）。本期保留 `setLevelPath` 遍历 `components` 树的语义，但改为**返回新的 components 树**（不可变），由 dispatch 一个 `setComponents` action 落库。**不经过 byId**。

新：

```js
/**
 * 遍历 components 树计算每个节点的 drillDownLevel
 * 返回新的 components 树（不修改原对象）
 * 同时遍历 node.children 和 node.data.config.drillDown
 *
 * 注意：level 是按树结构计算的（根=0，子=1，孙=2...）
 * drillDown 子树的 level 继承自父节点
 */
export function setLevelPath(nodes, parentNode, key = 'drillDownLevel') {
    if (!nodes || nodes.length === 0) return nodes;
    return nodes.map((node) => {
        const level = parentNode ? parentNode.data[key] + 1 : 0;
        const newNode = {
            ...node,
            data: {
                ...node.data,
                [key]: level,
            },
        };
        // 递归 children
        if (node.children && node.children.length > 0) {
            newNode.children = setLevelPath(node.children, newNode, key);
        }
        // 递归 drillDown（轮播子组件的层级数据）
        if (node.data.config?.drillDown && node.data.config.drillDown.length > 0) {
            newNode.data.config = {
                ...newNode.data.config,
                drillDown: setLevelPath(node.data.config.drillDown, newNode, key),
            };
        }
        return newNode;
    });
}
```

调用方改为：

```ts
// DesignerContent.tsx 的 onValueChange
const onValueChange = usePersistFn((uniqueId, value, level = 0) => {
    // setLevelPath 返回新树（不可变）
    const leveledComponents = setLevelPath(state.components, null);
    // mergeFieldConfig 在新树上工作
    const results = mergeFieldConfig(leveledComponents, { parentId: uniqueId, level }, value);
    // dispatch 替换整树（buildIndex 自动重建 byId）
    dispatch(setComponents(results));
});
```

**注意**：`setLevelPath` 改为返回新树后，所有调用方（`useDebounceMergeConfig` / `lock` / `visible` / `DesignerContent.onValueChange`）都要用返回值，不能再依赖"原地修改"。

**调研确认**：`drillDown level > 0` 实际无活跃调用方（grep `onValueChange.*,\s*[1-9]` 仅命中 `.bak` 文件），但保留 `level` 参数兼容性。`setLevelPath` 仍需处理 drillDown 子树，因为 `setLevelPath` 内部递归 `data.config.drillDown`。

#### ⚠️ realtimeDataFlow.record 的业务条件逻辑保留在 hook 层

现状 [DataProvider.tsx#L138-158](src/designer/DataProvider.tsx#L138) 的 `realtimeDataFlow.record` 内部有 `condition.withCondition(!params.enable, ...)` 分支：
- `params.enable === false` → `draft.splice(preIndex, 1)`（删除）
- `params.enable === true` → `Object.assign(draft[preIndex], {...})`（更新）

**reducer 不应承载这种业务条件**，应该由 hook 层判断后 dispatch 不同 action：

```ts
export const useRealtimeDataFlow = () => {
    const dispatch = useDispatch();
    return useMemo(() => ({
        record: (uniqueId, params) => {
            if (_.isNil(params.sourceId) || params.sourceId === '') return;
            // 业务条件判断留在 hook 层
            if (params.enable === false) {
                dispatch(deleteRealtimeDataFlow(uniqueId));
            } else {
                dispatch(recordRealtimeDataFlow({ uniqueId, params }));
            }
        },
        del: (uniqueId) => dispatch(deleteRealtimeDataFlow(uniqueId)),
    }), [dispatch]);
};
```

reducer 只做纯数据更新：

```ts
case 'designerCanvas/recordRealtimeDataFlow': {

        const preIndex = draft.realtimeDataFlow.findIndex((d) => d.uniqueId === uniqueId);

            Object.assign(draft.realtimeDataFlow[preIndex], { uniqueId, sourceId: params.sourceId });
```

### 3.6 useFlatComponents 改造

#### 现状

[Designer.tsx#L29-55](src/designer/common/context/context-designer/Designer.tsx#L29) 的 `useFlatComponents`：
- 内部调用 `useDesigner()` 读 `state.components`
- 用 `flatDesignerList(state.components)` 扁平化
- 用 `useLazyUpdate` 做懒更新
- 外部调用方：[search-layer.tsx#L17](src/designer/canvas-graph/components/search-layer.tsx#L17)、[layers-tree/tree/index.tsx#L28](src/designer/aside-panel/layers-tree/tree/index.tsx#L28)

#### 改造方案：改为订阅 byId 的 `useFlatFields`

`flatDesignerList` 的产出（一维数组 + `parentUniqueId` 链）和 `buildIndex` 的产出（byId + parentMap）**信息等价**。直接订阅 byId：

```ts
// src/store/designer/hooks.ts
export const useFlatFields = () => {
    return useSelector((s: RootReducerState) => {
        const { byId, parentMap } = s.designerCanvas;
        // 由 byId + parentMap 推导一维数组（保持原 flatDesignerList 的顺序：深度优先）
        // 注意：byId 不保证顺序，需要从 components 树遍历
        return s.designerCanvas.components; // 调用方自行 flatDesignerList 或改用 byId
    }, shallowEqual);
};
```

**问题**：`flatDesignerList` 返回的数组**保留了深度优先顺序**（用于 layers-tree 渲染顺序），而 `byId` 是无序 Map。直接订阅 byId 会丢失顺序。

**决策**：保留 `flatDesignerList` 函数（不删），但 `useFlatComponents` 改为订阅 `components` 树 + `useMemo` 重新扁平化：

```ts
// src/store/designer/hooks.ts
import { flatDesignerList } from '@Src/designer/renderer/utils';

export const useFlatComponents = () => {
    const components = useSelector(
        (s: RootReducerState) => s.designerCanvas.components,
        shallowEqual,
    );
    const flatComponents = useMemo(() => flatDesignerList(components), [components]);
    return [flatComponents] as const;
};
```

**注意**：`flatDesignerList` 内部的 `_.cloneDeep(curr)`（[utils.js#L502](src/designer/renderer/utils.js#L502)）在 task-009 §3.1 删除——但 `flatDesignerList` 只被 `useFlatComponents` 调用，且 `useMemo` 保证只在 components 引用变化时重算，cloneDeep 可安全去掉（调用方只读，不 mutate）。

#### 更激进的替代方案（可选，推荐）

当前方案订阅 `components` 树 + `useMemo` 重新扁平化，仍有 O(n) 计算成本。**更优方案**：

```ts
// 直接订阅 byId，返回有序数组
export const useFlatFields = () => {
    return useSelector((s: RootReducerState) => {
        const { byId, parentMap, components } = s.designerCanvas;
        // 由 components 树遍历生成深度优先有序数组（与原 flatDesignerList 顺序一致）
        // components 引用变化时才重算，byId 变化但 components 不变时（如 updateFieldConfig）跳过
        return flatDesignerList(components);
    }, shallowEqual);
};
```

**优势**：
- `useFlatFields` 订阅更精细（不强制订阅 components 数组）
- 但 `flatDesignerList` 依赖 components 遍历，实际等价

**结论**：当前方案（订阅 components + useMemo 重新扁平化）已经足够；激进方案无明显收益，**保留当前方案**。但保留 `useFlatFields` 接口作为未来扩展点。

#### 迁移路径

1. `Designer.tsx` 删除 `useFlatComponents` 定义，改为从 `@Src/store/designer/hooks` 导入
2. `Designer.tsx` 删除 `useDesigner` 定义
3. `search-layer.tsx` / `layers-tree/tree/index.tsx` 的 import 路径从 `@Src/designer/common` 改为 `@Src/store/designer/hooks`
4. `flatDesignerList` 函数保留在 `utils.js`，但去掉内部 cloneDeep
