# 物料开发上下文声明

```typescript
interface MaterialProps {
  // 基础属性
  uniqueId: string;
  type: string;
  className?: string;

  // 配置属性（来自 value 展开）
  config?: Record<string, any>;

  // 数据属性
  dataSource: any[];
  extraResponse?: any;

  // 设计器实例
  designer: {
    env: any;
    mode: 'development' | 'preview';
    container: HTMLElement;

    // 常量配置
    constants: {
      isMicroApp: boolean;
      isEnvProduction: boolean;
      BASE_NAME: string;
      STATIC_PATH: string;
      WORKER_PATH: string;
      IMAGE_PATH: string;
      PUBLIC_URL: string;
      LOCALES_PATH: string;
      MICRO_APP_URL: string;
      CONFIG_HOST: string;
      CONFIG_LOCAL: string;
      PROJECT_NAME: string;
      DEV_ENABLE_LOCAL_ENV: boolean;
      ENABLE_LOCAL_DESIGNER: boolean;
      NODE_ENV_DEVELOPMENT: boolean;
      SDK_PATH: string;
      META_HUMAN_SDK_PATH: string;
    };

    prefix: { dynamicEventPrefix: string; };
    permissions: {
      zoneName?: any;
      zoneLevel?: any;
      zoneId?: any;
      userId?: any;
      currentLoginInfo?: any;
    };
    fieldName: { DRILL_DOWN: string; };
    utils: {
      logger: any;
      getDesignerOriginData: (data: any) => any;
      eachRequestParams: (params: any, callback: any) => void;
      parseSubscribeParams: (value: string) => {
        subscribeKey: string | null;
        isSubscribeKey: boolean;
      };
      navigateTo: (to: string, opts?: { type?: 'push' | 'replace'; state?: any }) => void;
      history: History;
    };
    api: {
      customDataSourceApi: (type: 'api' | 'dataSet', params: any) => Promise<any>;
      request: (...args: any[]) => Promise<any>;
      buildCustomApiParams: (params: any, paramsMapping: any, callback?: any) => {
        params: any;
        hasInvalidateParams: boolean;
      };
    };
  };

  // 交互API
  interaction: {
    dispatch: (args: {
      type?: string;
      data: Array<{ fieldName: string; state: any }>;
    }) => void;
    dispatchRealtimeDataFlow: (list: any[]) => void;
    subscribe: Array<{
      fieldName: string | number | null;
      uniqueId: string | number | null;
      state: any;
      time: number;
      specVersion?: string | null;
      type?: string | null;
      dataContentType?: string;
      dataSchema?: any;
      data?: any;
    }> | null;
    defined: Record<string, any>;
    action: Map<string, any>;
  };

  // 交互参数
  interactionProps: Record<string, any>;

  // URL参数
  receivedPropsParams: Record<string, any>;
}
```

## 使用示例

```typescript
const MyMaterial = (props: MaterialProps) => {
    const { uniqueId, type, dataSource, config, designer, interaction, interactionProps } = props;

    // 派发交互事件
    const handleClick = () => {
        interaction.dispatch({
            data: [{ fieldName: 'myEvent', state: { value: 'clicked' } }]
        });
    };

    // 使用订阅数据
    const subscribeValue = interactionProps.mySubscribeField;

    // 路由导航
    designer.utils.navigateTo('/new-page');

    // 日志记录
    designer.utils.logger.log('rendered');

    // 使用常量
    const isProd = designer.constants.isEnvProduction;
    const staticPath = designer.constants.STATIC_PATH;

    return (
        <div id={uniqueId}>
            <h1>{type}</h1>
            <button onClick={handleClick}>触发事件</button>
            <div>订阅值: {subscribeValue}</div>
            <ul>{dataSource.map((item, index) => <li key={index}>{item.name}</li>)}</ul>
        </div>
    );
};
```

## 注意事项

1. `value` 的子属性有哪些，取决于物料在编辑器中的配置，物料开发人员应通过实际调试确认。

2. `animation` 配置由编辑器内部的 `AnimationPlugin.FieldAnimation` 组件消费，物料不需要处理。

3. `designer.mode` 可判断当前是开发模式还是预览模式。

4. `designer.utils.getDesignerOriginData(data)` 可获取原始数据。
