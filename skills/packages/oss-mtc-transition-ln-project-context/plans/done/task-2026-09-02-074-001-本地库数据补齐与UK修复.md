### 3.4 导入脚本生成器（可直接落盘使用）

> 放到 `backend/_tmp_gen_import_sql.py`，在 `backend/` 下用 `.venv\Scripts\python.exe` 执行，产物写入 `backend/scripts/test-ln-db/import_reexport_bigscreen.sql`；生成后**先 review SQL 再执行**（执行方式：psql / DBeaver / 一次性 Python 均可，按分号拆语句逐条跑）。用完删除临时脚本（仓库不留）。

```python
"""从 PM reexport CSV 生成大屏表导入 SQL。

- letter_screen_1：整表重建，保留 CSV 真实 ID
- stats_jdlk_*：整表重建，ID = 行内容 md5（CSV 的 ID 是导出固定值，非真主键）
"""
import csv
import hashlib
import sys
from pathlib import Path

import psycopg2

sys.path.insert(0, ".")
from app.core.config import settings  # noqa: E402

REPO = Path(__file__).resolve().parents[1]
CSV_DIR = REPO / "docs/research/db/数据库测试结果-20260827-pm/59.197.182.105_reexport"
REBUILD_TABLES = [
    "letter_screen_1",
    "stats_jdlk_persion_age_group",
    "stats_jdlk_persion_employment",
    "stats_jdlk_persion_hjlb",
    "stats_jdlk_persion_infor_cant",
]
OUT = REPO / "backend/scripts/test-ln-db/import_reexport_bigscreen.sql"


def lit(v: str) -> str:
    v = (v or "").strip()
    return "NULL" if v == "" else "'" + v.replace("'", "''") + "'"


url = settings.database_url
body = url.split("://", 1)[1]
auth, hostdb = body.rsplit("@", 1)
user, pwd = auth.split(":", 1)
host, port = hostdb.split(":")[0], hostdb.split(":")[1].split("/")[0]
db = hostdb.split("/")[-1].split("?")[0]
conn = psycopg2.connect(host=host, port=port, user=user, password=pwd, dbname=db)
cur = conn.cursor()

parts = [
    "-- 大屏表数据补齐：PM reexport CSV -> Kingbase dw_basic_lc",
    "-- 生成于 task-2026-09-02-074-001，重跑前先执行 DROP UK（见 task §3.3）",
    "",
]
for t in REBUILD_TABLES:
    cur.execute(
        "SELECT column_name FROM information_schema.columns "
        "WHERE table_schema='dw_basic_lc' AND table_name=%(t)s ORDER BY ordinal_position",
        {"t": t},
    )
    db_cols = [r[0] for r in cur.fetchall()]
    with open(CSV_DIR / f"{t}.csv", encoding="utf-8-sig", newline="") as f:
        rows = list(csv.DictReader(f))
    col_map = {c.upper(): c for c in db_cols}
    cols = [col_map[c] for c in rows[0].keys() if c.upper() in col_map and c.upper() != "ID"]
    all_cols = ["ID"] + cols
    parts.append(f"-- ===== {t}（{len(rows)} 行，整表重建）=====")
    parts.append("BEGIN;")
    parts.append(f'DELETE FROM dw_basic_lc."{t}";')
    seen: set[str] = set()
    values = []
    for r in rows:
        id_val = r["ID"] if t == "letter_screen_1" else hashlib.md5(
            "\x01".join((r.get(c) or "").strip() for c in cols).encode("utf-8")
        ).hexdigest()
        if id_val in seen:
            continue
        seen.add(id_val)
        values.append("(" + ", ".join([lit(id_val)] + [lit(r.get(c, "")) for c in cols]) + ")")
    col_list = ", ".join(f'"{c}"' for c in all_cols)
    for i in range(0, len(values), 500):
        parts.append(
            f'INSERT INTO dw_basic_lc."{t}" ({col_list}) VALUES\n'
            + ",\n".join(values[i : i + 500]) + ";"
        )
    parts.append("COMMIT;")
    parts.append("")

conn.close()
OUT.parent.mkdir(parents=True, exist_ok=True)
OUT.write_text("\n".join(parts), encoding="utf-8")
print("written:", OUT)
```

### 3.5 关键坑清单（重跑前必读，每条都是实测踩过）

1. `letter_screen_1` 的 UK `uk_letter_scr1_org_auth_dt (ORGAN_ID, REG_AUTHORITY, STATS_DATE)` 与真实数据冲突（同一键下每诉求分类 1 行）——**必须先 §3.3 DROP**，否则批量插入报唯一冲突
2. `stats_jdlk_*` 的 CSV `ID` 列是导出固定值（如 age_group 6572 行只有 4 个不同 ID），**不能**当主键 upsert，必须 md5 重新生成
3. `letter_screen_1` 旧样例（268 行）与 reexport（992 行）ID **零重叠**——不能增量 upsert，必须整表 DELETE + INSERT，否则 SUM 双倍计数
4. CSV 编码是 `utf-8-sig`（带 BOM），列名大小写要与本地库列做 `upper()` 对齐映射
5. `aa_jycy.csv` 为空文件（0 行），跳过；`letter_screen_2/3/5/6` 本地已是全量（673/673/270/270），无需重导
6. 生成 SQL 时值转义：空串 → `NULL`，单引号双写；按 500 行/语句分批
